\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/pipeline.png}
  \caption{MemeSonic system overview. \textbf{Phase 1} generates expressive meme voiceover by
  conditioning ElevenLabs TTS on a GPT-4.1-mini script and a Qwen3-based prosody prompt, with
  prosody further modulated by the incongruity score $\delta$. \textbf{Phase 2} computes $\delta$
  from per-modality emotion distributions and routes each sample through FusionMoE (Expert~R:
  Concat, Expert~U: Late Fusion, Expert~S: Incongruity Fusion). \textbf{Phase 3} aligns
  Emotion2Vec speech embeddings into the shared CLIP/ImageBind space via a linear projector and
  fuses all three modalities for classification.}
  \label{fig:pipeline}
\end{figure}
